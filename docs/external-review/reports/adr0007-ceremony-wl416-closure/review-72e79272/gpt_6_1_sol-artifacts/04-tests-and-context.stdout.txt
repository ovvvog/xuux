    84	function childExitedOrZombie(pid) {
    85	  try {
    86	    const st = readFileSync(`/proc/${pid}/stat`, 'utf8');
    87	    return st.slice(st.lastIndexOf(')') + 2).split(' ')[0] === 'Z';
    88	  } catch {
    89	    return true;
    90	  }
    91	}
    92	afterEach(async () => {
    93	  for (const c of trackedChildren) {
    94	    if (c.exitCode !== null || c.signalCode !== null) {
    95	      trackedChildren.delete(c);
    96	      continue;
    97	    }
    98	    if (!c.pid || !childExitedOrZombie(c.pid)) continue; // ما زالَ حيًّا: لا يُنتظرُ ولا يُقتَل
    99	    await Promise.race([

   104	  }
