   530	    // بدأتْها. فحوصُ الردودِ التي تعتمدُ على حالةٍ سابقةٍ (اسمُ الأصلِ في بياناتِ الإنشاءِ، وتذكرةُ PolicySigned
   531	    // الحقيقيّةُ عندَ expiration سالبٍ) تُربَطُ به؛ مقبضٌ غيرُ مسجّلٍ ⇒ فشلٌ مغلقٌ مسمّى لا قبولٌ بنيويٌّ.
   532	    public sealed class ResponseContext
   533	    {
   534	        private readonly Dictionary<uint, byte[]> _names = new Dictionary<uint, byte[]>();
   535	        private readonly Dictionary<uint, byte[]> _qualified = new Dictionary<uint, byte[]>();
   536	        private readonly Dictionary<uint, uint> _hierarchy = new Dictionary<uint, uint>();
   537	        private readonly Dictionary<uint, byte> _sessionType = new Dictionary<uint, byte>();

   539	        public bool TryObject(uint h, out byte[] name, out byte[] qualifiedName, out uint hierarchy)
   540	        {
   541	            name = null; qualifiedName = null; hierarchy = 0;
   542	            if (!_names.ContainsKey(h)) return false;
   543	            name = _names[h]; qualifiedName = _qualified[h]; hierarchy = _hierarchy[h];
   544	            return true;
   545	        }
   546	
   547	        public bool TrySessionType(uint h, out byte type)
   548	        {
   549	            type = 0;
   550	            if (!_sessionType.ContainsKey(h)) return false;
   551	            type = _sessionType[h];
   552	            return true;
   553	        }
   554	
   555	        // هل بقيَ للعمليّةِ سياقٌ مسجَّلٌ؟ — شاهدُ الحالةِ الفعليّةِ لا مجاوَرةِ النداءِ: القيمةُ تُشتَقُّ من
   556	        // الخرائطِ نفسِها بعدَ استدعاءاتِ الإزالةِ، فحذفُ EndOfLife مع إبقاءِ الحقيقةِ يُسقطُ الشاهدَ (جولةُ 4).
   557	        public bool Knows(uint h)
   558	        {
   559	            return _names.ContainsKey(h) || _sessionType.ContainsKey(h);
   560	        }
   561	
   562	        // الاسمُ المؤهَّلُ (Part 1 §16): QN(B) = nameAlg || H(QN(أصلِ B) || Name(B))، وQN للهرميّةِ = مقبضُها (4 بايتات).
   563	        // قياسُ swtpm 2026-10-18: parentQualifiedName في ردِّ Create = 000B || SHA256(40000001 || Name(primary)).
   564	        public static byte[] Qualified(byte[] parentQn, byte[] name)
   565	        {
   566	            byte[] cat = new byte[parentQn.Length + name.Length];
   567	            Array.Copy(parentQn, 0, cat, 0, parentQn.Length);
   568	            Array.Copy(name, 0, cat, parentQn.Length, name.Length);
   569	            byte[] d = Sha.H(cat);
   570	            byte[] q = new byte[2 + d.Length];
   571	            q[0] = 0x00; q[1] = 0x0B;
   572	            Array.Copy(d, 0, q, 2, d.Length);
   573	            return q;
   574	        }
   575	
   576	        private static uint U32At(byte[] b, int o) { return ((uint)b[o] << 24) | ((uint)b[o + 1] << 16) | ((uint)b[o + 2] << 8) | b[o + 3]; }
   577	
   578	        public static byte[] HierarchyName(uint h) { return new byte[] { (byte)(h >> 24), (byte)(h >> 16), (byte)(h >> 8), (byte)h }; }
   579	
   580	        // يُستدعى بعدَ ResponseSchema.Check الناجحِ فقط (ردُّ رمزِه صفرٌ ومخطّطُه صالح).
   581	        public void Record(uint cc, byte[] cmd, uint[] handles, byte[] prm)
   582	        {
   583	            switch (cc)
   584	            {
   585	                case Tpm.CC_CreatePrimary:
   586	                {
   587	                    uint hier = U32At(cmd, 10);
   588	                    Rd r = new Rd(prm); r.B2B(); r.B2B(); r.B2B(); r.U16(); r.U32(); r.B2B();
   589	                    byte[] nm = r.B2B();
   590	                    Put(handles[0], nm, Qualified(HierarchyName(hier), nm), hier);
   591	                    return;
   592	                }
   593	                case Tpm.CC_Load:
   594	                {
   595	                    uint parent = U32At(cmd, 10);
   596	                    byte[] pn, pq; uint ph;
   597	                    if (!TryObject(parent, out pn, out pq, out ph)) return; // أصلٌ غيرُ معروفٍ ⇒ لا يُسجَّلُ الابنُ فتُرفضُ فحوصُه لاحقًا
   598	                    byte[] nm = new Rd(prm).B2B();
   599	                    Put(handles[0], nm, Qualified(pq, nm), ph);
   600	                    return;
   601	                }
   602	                case Tpm.CC_StartAuthSession:
   603	                {
   604	                    Rd c = new Rd(cmd); c.Raw(10 + 8); c.B2B(); c.B2B();
   605	                    // مقبضٌ كان كائنًا ثمّ عادَ جلسةً: سياقُ الكائنِ لا يُورَّثُ للجلسةِ (WL-416) — دورةُ حياةِ
   606	                    // السياقِ تتبعُ دورةَ حياةِ المقبضِ، والعكسُ موجودٌ سلفًا في Put (إزالةُ نوعِ الجلسةِ).
   607	                    _names.Remove(handles[0]); _qualified.Remove(handles[0]); _hierarchy.Remove(handles[0]);
   608	                    _sessionType[handles[0]] = c.U8();
   609	                    return;
   610	                }
   611	                case Tpm.CC_FlushContext:
   612	                {
   613	                    uint h = U32At(cmd, 10);
   614	                    _names.Remove(h); _qualified.Remove(h); _hierarchy.Remove(h); _sessionType.Remove(h);
   615	                    return;
   616	                }
   617	            }
   618	        }

   620	        private void Put(uint h, byte[] name, byte[] qn, uint hier)
   621	        {
   622	            _names[h] = name; _qualified[h] = qn; _hierarchy[h] = hier; _sessionType.Remove(h);
   623	        }
   624	
   625	        // ── إزالةُ السياقِ عندَ إثباتِ انتهاءِ العمرِ (WL-416) ──
   626	        // تُربطُ بالحدثِ الذي يُثبتُ زوالَ المقبضِ أو الجلسةِ فعلًا، لا بمجردِ نتيجةٍ مجهولةٍ:
   627	        //   1. نجاحُ FlushContext المؤكَّد (عبرَ Record عندَ رمزِ صفرٍ).
   628	        //   2. جردٌ قرائيٌّ يُثبتُ غيابَ المقبضِ بعدَ فقدِ ردِّ التحريرِ (مسارُ المصالحةِ).
   629	        //   3. نجاحُ العمليةِ التي تستهلكُ الجلسةَ (جلسةُ السياسةِ تُستهلَكُ بأمرِ التفويضِ الناجحِ —
   630	        //      سلوكُ المحاكي المُثبَتُ: الجردُ النهائيّ لا يُبقي جلساتٍ مستعملةً).
   631	        // المقبضُ الحيّ أو الجردُ غيرُ المحسومِ لا يُزيلُ السياقَ — الفشلُ المغلقُ باقٍ، والسياقُ المفقودُ لا
   632	        // يتحوّلُ قبولًا افتراضيًّا: فحوصُ الكائناتِ والجلساتِ المجهولةِ تظلُّ مسمّاةً (CONTEXT_UNKNOWN).
   633	        public void EndOfLife(uint h)
   634	        {
   635	            _names.Remove(h); _qualified.Remove(h); _hierarchy.Remove(h); _sessionType.Remove(h);
   636	        }
