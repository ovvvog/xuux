    24	export function assertSimulatorTcti(tcti) {
    25	  const value = String(tcti ?? '').trim();
    26	  if (value.length === 0) {
    27	    throw new Error('TCTI_REAL_OR_UNKNOWN: empty TCTI');
    28	  }
    29	  if (/device:/.test(value) || /\/dev\/tpm/.test(value)) {
    30	    throw new Error('TCTI_REAL_OR_UNKNOWN: device TCTI refused: ' + value);
    31	  }
    32	  // رفض أي محارف shell خطرة في سلسلة TCTI (حقن).
    33	  if (/[^A-Za-z0-9:=,.\-_]/.test(value)) {
    34	    throw new Error('TCTI_REAL_OR_UNKNOWN: unsafe characters in TCTI: ' + value);
    35	  }
    36	  if (!SIM_TCTI_PATTERNS.some((re) => re.test(value))) {
    37	    throw new Error('TCTI_REAL_OR_UNKNOWN: not a simulator TCTI: ' + value);
    38	  }
    39	  return value;
    40	}
