    32	        public BlockingTransport(ICeremonyTransport inner, uint commandCode, int nth, bool all)
    33	        {
    34	            _inner = inner; _cc = commandCode; _nth = nth; _all = all;

    39	        public byte[] Transmit(byte[] command)
    40	        {
    41	            uint cc = ((uint)command[6] << 24) | ((uint)command[7] << 16) | ((uint)command[8] << 8) | command[9];
    42	            if (cc == _cc)
    43	            {
    44	                _seen++;
    45	                if (_all || _seen == _nth)
    46	                {
    47	                    BlockedCount++;
    48	                    return new byte[] { 0x80, 0x01, 0, 0, 0, 10, 0x80, 0x28, 0x04, 0x00 };
    49	                }
    50	            }
    51	            Forwarded.Add(Hex.U32(cc));
    52	            return _inner.Transmit(command);
    53	        }
    54	    }

   106	        // جلسةُ فخٍّ تنشأُ قبلَ بدءِ البروفةِ (وضعُ session-alias): مقبضٌ موجودٌ في خطِّ الأساسِ يُعادُ في ردِّ
   107	        // StartAuthSession فيُرفَضُ تبنّيه (RESPONSE_HANDLE_PREEXISTING) — إعادةُ إنتاجِ SOL413-01.
   108	        public CeremonyFaultTransport(ICeremonyTransport inner, string mode, bool decoySession)
   109	        {
   110	            _inner = inner; _mode = mode;
   111	            if (decoySession)
   112	            {
   113	                byte[] z = new byte[32]; new System.Random(4).NextBytes(z);
   114	                Resp p = Resp.Parse(inner.Transmit(Cmd.StartAuthSession(Tpm.SE_POLICY, z)), 1, Tpm.ST_NO_SESSIONS, 0, new bool[0]);
   115	                _decoySession = p.Handles[0];
   116	            }
   117	            // WL-416: زرعُ فهرسٍ حقيقيّ قبلَ البروفةِ (inventory-more0-hide خارجَ مجالِ المالكِ؛
   118	            // nv-preseed-unattributable داخلَهُ) — التعريفُ من النقلِ لا من خطواتِ البروفةِ.
   119	            if (mode == "inventory-more0-hide") _plantedNv = 0x01000001;
   120	            if (mode == "nv-preseed-unattributable") _plantedNv = 0x01807777;
   121	            if (_plantedNv != 0)
   122	            {
   123	                Resp d = Resp.Parse(inner.Transmit(Cmd.NvDefineSpace(new byte[0], new byte[0], PlantedNvPublic(_plantedNv))), 0, Tpm.ST_SESSIONS, 1, new bool[] { true });
   124	                if (d.Rc != 0) throw new InvalidOperationException("planted NV define failed rc=" + Hex.U32(d.Rc));
   125	            }
   126	        }

   131	        private static byte[] Append(byte[] raw, byte[] extra)
   132	        {

   140	        // يعيدُ كتابةَ منطقةِ تفويضِ ردِّ ST_SESSIONS (مسابرُ الدفعةِ الثالثةِ): البنيةُ الكلّيّةُ متّسقةٌ والمحتوى

   185	        public byte[] Transmit(byte[] command)
   186	        {
   187	            uint cc = Cc(command);
   188	            Forwarded.Add(Hex.U32(cc));
   189	            if (_mode == "flush-refusal" && cc == Tpm.CC_FlushContext)
   190	            {
   191	                // رفضٌ مستمرٌّ لكلِّ تحريرٍ (لا الأولى فقط): الجلسةُ يجبُ أن تبقىَ معلَنةً في البقايا.
   192	                if (Changes == 0) { Rd c = new Rd(command); c.U16(); c.U32(); c.U32(); LostSession = c.U32(); }
   193	                Changes++;
   194	                return RcOnly(0x80280400);
   195	            }
   196	            if (_mode == "unexpected-negative" && cc == Tpm.CC_PolicySigned)
   197	            {
   198	                // الثالثةُ هي رفضُ إعادةِ التوقيعِ: يُعادُ رمزٌ عامٌّ (0x101) مكانَ رفضِ التوقيعِ المقصودِ.
   199	                Changes++;
   200	                if (Changes == 3) return RcOnly(0x101);
   201	            }
   202	            if (_mode == "blocked-forever-undefine" && cc == Tpm.CC_NV_UndefineSpace)
   203	            {
   204	                Changes++;
   205	                return RcOnly(Tpm.TPM_E_COMMAND_BLOCKED);
   206	            }
   207	            if (_mode == "define-not-executed" && cc == Tpm.CC_NV_DefineSpace && Changes == 0)
   208	            {
   209	                Changes++;
   210	                throw new System.IO.IOException("WL413_DEFINE_DROPPED_BEFORE_EXECUTION");
   211	            }
   212	            if (_mode == "identity-string34" && cc == Tpm.CC_GetCapability && Changes == 0)
   213	            {
   214	                Changes++;
   215	                Wr p = new Wr().U8(0).U32(6).U32(5)
   216	                    .U32(0x105).U32(0x49424D00)   // "IBM"
   217	                    .U32(0x106).U32(0x53572020)   // "SW  "
   218	                    .U32(0x107).U32(0x2054504D)   // " TPM"
   219	                    .U32(0x108).U32(0x4556494C)   // "EVIL"
   220	                    .U32(0x109).U32(0x58585858);  // "XXXX"
   221	                byte[] body = p.Bytes();
   222	                return new Wr().U16(Tpm.ST_NO_SESSIONS).U32((uint)(10 + body.Length)).U32(0).Raw(body).Bytes();
   223	            }
   224	            if (_mode == "identity-spacefold" && cc == Tpm.CC_GetCapability && Changes == 0)
   225	            {
   226	                Changes++;
   227	                Wr p = new Wr().U8(0).U32(6).U32(5)
   228	                    .U32(0x105).U32(0x49424D00)
   229	                    .U32(0x106).U32(0x53205720)   // "S W "
   230	                    .U32(0x107).U32(0x5420504D)   // "T PM"
   231	                    .U32(0x108).U32(0).U32(0x109).U32(0);
   232	                byte[] body = p.Bytes();
   233	                return new Wr().U16(Tpm.ST_NO_SESSIONS).U32((uint)(10 + body.Length)).U32(0).Raw(body).Bytes();
   234	            }
   235	            if (_mode == "identity-missing-prop" && cc == Tpm.CC_GetCapability && Changes == 0)
   236	            {
   237	                Changes++;
   238	                // أربعُ خصائصٍ فقط: سلسلةُ البائعِ الرابعةُ غائبةٌ — الهويّةُ ناقصةٌ فتُرفض.
   239	                Wr p = new Wr().U8(0).U32(6).U32(4)
   240	                    .U32(0x105).U32(0x49424D00)
   241	                    .U32(0x106).U32(0x53572020)
   242	                    .U32(0x107).U32(0x2054504D)
   243	                    .U32(0x108).U32(0);
   244	                byte[] body = p.Bytes();
   245	                return new Wr().U16(Tpm.ST_NO_SESSIONS).U32((uint)(10 + body.Length)).U32(0).Raw(body).Bytes();
   246	            }
   247	            if (_dead) throw new System.IO.IOException("WL413R2_TRANSPORT_DEAD");
   248	            // ── مسابرُ الدفعةِ الثالثةِ (WL-413 R3) ──
   249	            if (_mode == "written-before-reconcile" && cc == Tpm.CC_NV_DefineSpace && Changes == 0)
   250	            {
   251	                // يُنفَّذُ التعريفُ، ثمّ «طرفٌ آخرُ» يكتبُ الفهرسَ (زيادةٌ بتفويضٍ فارغٍ يسمحُ بها شكلُ S1)،
   252	                // ثمّ يضيعُ ردُّ التعريف. الفهرسُ يطابقُ تعريفَنا عدا WRITTEN — ليس مُثبَتَ الملكيّة.
   253	                Rd c = new Rd(command); c.U16(); c.U32(); c.U32(); c.U32(); uint authSize = c.U32(); c.Raw((int)authSize); c.B2B();
   254	                DefinedPublic = c.B2B(); DefinedHandle = new Rd(DefinedPublic).U32();
   255	                _inner.Transmit(command);
   256	                byte[] inc = _inner.Transmit(Cmd.NvIncrement(DefinedHandle, Cmd.PasswordAuth(new byte[0])));
   257	                Rd ih = new Rd(inc); ih.U16(); ih.U32(); ThirdPartyRc = ih.U32();
   258	                Changes++;
   259	                throw new System.IO.IOException("WL413R2_RESPONSE_LOSS_AFTER_DEFINE_AND_THIRD_PARTY_WRITE");
   260	            }
   261	            if (_mode == "neg-increment-loss" && cc == Tpm.CC_NV_Increment)
   262	            {
   263	                // الزيادةُ الرابعةُ هي s2v:NV_Increment(password-auth) — مرفوضةٌ على المحاكي، وردُّها يضيع.
   264	                _increments++;
   265	                if (_increments == 4) { _inner.Transmit(command); Changes++; throw new System.IO.IOException("WL413R2_LOSS_AFTER_NEGATIVE_INCREMENT"); }
   266	            }
   267	            if (_mode == "transport-dead-after-define" && cc == Tpm.CC_NV_DefineSpace && Changes == 0)
   268	            {
   269	                Rd c = new Rd(command); c.U16(); c.U32(); c.U32(); c.U32(); uint authSize = c.U32(); c.Raw((int)authSize); c.B2B();
   270	                DefinedPublic = c.B2B(); DefinedHandle = new Rd(DefinedPublic).U32();
   271	                _inner.Transmit(command); Changes++; _dead = true;
   272	                throw new System.IO.IOException("WL413R2_TRANSPORT_DIED_AFTER_DEFINE");
   273	            }
   274	            if (_mode == "create-response-loss" && cc == Tpm.CC_Create && Changes == 0)
   275	            {
   276	                Changes++; _inner.Transmit(command);
   277	                throw new System.IO.IOException("WL413R3_CREATE_RESPONSE_LOSS");
   278	            }
   279	            if (_mode == "flush-response-loss-noexec" && cc == Tpm.CC_FlushContext)
   280	            {
   281	                // الردُّ يضيعُ والأمرُ لم يُنفَّذ أصلًا: نتيجةٌ مجهولةٌ والمقبضُ باقٍ بحقٍّ — المصالحةُ القرائيّةُ
   282	                // تُبقي السياقَ ولا تدّعي زوالًا (الزوجُ الصادقُ لـ flush-response-loss — حسمُ المراجعةِ 3).
   283	                Changes++;
   284	                throw new System.IO.IOException("WL416_FLUSH_RESPONSE_LOSS_NOEXEC");
   285	            }
   286	            if (_mode == "flush-response-loss" && cc == Tpm.CC_FlushContext)
   287	            {
   288	                // الأمرُ يُنفَّذُ على الناقلِ الداخليّ لكنّ ردّهُ يضيعُ: نتيجةٌ مجهولةٌ تُفعّلُ المصالحةَ القرائيّةَ —
   289	                // الجردُ يُثبِتُ غيابَ المقبضِ فيُزالُ السياقُ من المسارِ الحقيقيّ (WL-416، حسمُ المراجعةِ 3).
   290	                Changes++; _inner.Transmit(command);
   291	                throw new System.IO.IOException("WL416_FLUSH_RESPONSE_LOSS");
   292	            }
   293	            if (_mode == "policysigned-neg-loss" && cc == Tpm.CC_PolicySigned)
   294	            {
   295	                // الرفضُ الثاني هو s2v.replay (old-signature) — مُغيّرٌ ومرفوضٌ ويضيعُ ردُّه (SOL413-06).
   296	                Changes++;
   297	                if (Changes == 3) { _inner.Transmit(command); throw new System.IO.IOException("WL413R3_NEGATIVE_POLICYSIGNED_LOSS"); }
   298	            }
   299	            if (_mode == "readcounter-malformed-14a" && cc == Tpm.CC_NV_Increment)
   300	            {
   301	                // الزيادةُ الرابعةُ هي s2v:NV_Increment(password-auth) — مرفوضةٌ على المحاكي ويضيعُ ردُّها،
   302	                // ثمّ يُشوّهُ ردُّ القراءةِ 0x14A في المصالحةِ — كان يُبتلَعُ دليلًا للعدمِ (SOL413-05).
   303	                _increments++;
   304	                if (_increments == 4) { _inner.Transmit(command); Changes++; throw new System.IO.IOException("WL413R3_NEG_INCREMENT_LOSS_THEN_MALFORMED_14A"); }
   305	            }

   900	    //   1. فقدُ ردِّ التحريرِ + إثباتُ الغيابِ بجردٍ قرائيّ + إعادةُ استخدامِ المقبضِ بلا تسجيلٍ (طرفٌ ثالثٍ)
   901	    //      ⇒ لا وراثةَ للسياقِ، ورفضٌ مسمّى (RESPONSE_CREATION_PARENT_UNKNOWN) لا ادّعاءُ تلفٍ.
   902	    //   2. إعادةُ الاستخدامِ معَ تسجيلٍ ⇒ مقبولٌ، والاسمُ الجديدُ لا يرثُ الاسمَ القديم.
   903	    //   3. فقدُ الردِّ والمقبضُ ما زالَ حاضرًا (جردٌ قرائيّ) ⇒ السياقُ محفوظٌ (لا EndOfLife).
   904	    //   4. استهلاكُ جلسةِ السياسةِ يُقاسُ مباشرةً: الجلسةُ تختفي فورَ الأمرِ المُفوِّزِ الناجحِ (قبلَ التنظيفِ
   905	    //      وقبلَ تحريرِ غيرِها) ثمّ EndOfLife يزيلُ نوعَها المسجّل.
   906	    // السياقُ يُستعمَلُ بالانعكاسِ — هذه أداةُ قياسٍ للاختبارِ لا مسارَ البروفةِ.
   907	    public static class ContextLifetimeProbe
   908	    {
   909	        private static uint U32(byte[] b, int o) { return ((uint)b[o] << 24) | ((uint)b[o + 1] << 16) | ((uint)b[o + 2] << 8) | b[o + 3]; }
   910	
   911	        private static byte[] Params(byte[] r, int handles)
   912	        {
   913	            int o = 10 + 4 * handles;
   914	            if (r[1] == 2) { int n = (int)U32(r, o); byte[] p = new byte[n]; Array.Copy(r, o + 4, p, 0, n); return p; }
   915	            byte[] q = new byte[r.Length - o]; Array.Copy(r, o, q, 0, q.Length); return q;
   916	        }
   917	
   918	        // جردٌ قرائيّ للمقابضِ (صفحاتُ 64) — إثباتُ حضورٍ/غيابٍ لا مساسَ بالحالةِ.
   919	        private static bool HandlePresent(ICeremonyTransport t, uint start, uint handle)
   920	        {
   921	            for (int page = 0; page < 16; page++)
   922	            {
   923	                byte[] r = t.Transmit(Cmd.GetCapability(1, start, 64));
   924	                if (U32(r, 6) != 0) throw new InvalidOperationException("inventory rc=" + Hex.U32(U32(r, 6)));
   925	                Rd q = new Rd(Params(r, 0)); byte more = q.U8(); q.U32(); uint n = q.U32(); uint last = 0;
   926	                for (uint i = 0; i < n; i++) { uint h = q.U32(); if (h == handle) return true; last = h; }
   927	                if (more == 0 || n == 0) return false;
   928	                start = last + 1;
   929	            }
   930	            throw new InvalidOperationException("inventory incomplete");
   931	        }
   932	
   933	        private static byte[] RequireOk(ICeremonyTransport t, byte[] cmd, string what)
   934	        {
   935	            byte[] r = t.Transmit(cmd);
   936	            if (U32(r, 6) != 0) throw new InvalidOperationException(what + " rc=" + Hex.U32(U32(r, 6)));
   937	            return r;
   938	        }

   940	        public static List<string> Run(ICeremonyTransport t)
   941	        {
   942	            List<string> lines = new List<string>();
   943	            Type ctxType = typeof(ResponseSchema).Assembly.GetType("Xuux.TpmCeremony.ResponseContext");
   944	            object ctx = ctxType == null ? null : Activator.CreateInstance(ctxType);
   945	            System.Reflection.MethodInfo record = ctxType == null ? null : ctxType.GetMethod("Record");
   946	            System.Reflection.MethodInfo endOfLife = ctxType == null ? null : ctxType.GetMethod("EndOfLife");
   947	            System.Reflection.MethodInfo tryObject = ctxType == null ? null : ctxType.GetMethod("TryObject");
   948	            System.Reflection.MethodInfo trySessionType = ctxType == null ? null : ctxType.GetMethod("TrySessionType");
   949	            System.Reflection.MethodInfo check = ctxType == null
   950	                ? typeof(ResponseSchema).GetMethod("Check", new Type[] { typeof(uint), typeof(byte[]), typeof(byte[]) })
   951	                : typeof(ResponseSchema).GetMethod("Check", new Type[] { typeof(uint), typeof(byte[]), typeof(byte[]), ctxType });
   952	            if (ctxType == null)
   953	            {
   954	                lines.Add("constraint: ResponseContext type absent on this baseline; lifetime cases not measurable here");
   955	                return lines;
   956	            }
   957	            // الحالةُ 1+2: فقدُ ردِّ التحريرِ، إثباتُ الغيابِ، إعادةُ الاستخدامِ (بلا تسجيلٍ ثمّ بهِ).
   958	            byte[] primaryCmd = Cmd.CreatePrimary(new byte[0], Templates.StoragePrimary());
   959	            byte[] primaryReply = RequireOk(t, primaryCmd, "primary");
   960	            uint primary = U32(primaryReply, 10);
   961	            record.Invoke(ctx, new object[] { Tpm.CC_CreatePrimary, primaryCmd, new uint[] { primary }, Params(primaryReply, 1) });
   962	            object[] oa = new object[] { primary, null, null, null };
   963	            bool known = (bool)tryObject.Invoke(ctx, oa);
   964	            byte[] oldName = (byte[])oa[1]; byte[] oldQn = (byte[])oa[2]; uint oldHier = (uint)oa[3];
   965	            lines.Add("record: known=" + known + " hier=" + Hex.U32(oldHier) + " nameLen=" + oldName.Length);
   966	            // فقدُ ردِّ التحريرِ: يُرسَلُ الأمرُ ويُهمَلُ الردُّ (لا تسجيلَ ولا فحصَ) — نتيجةٌ مجهولةٌ للبروفةِ.
   967	            RequireOk(t, Cmd.FlushContext(primary), "flush-loss");
   968	            bool presentAfterLoss = HandlePresent(t, 0x80000000, primary);
   969	            lines.Add("flush-loss: presentAfterLoss=" + presentAfterLoss);
   970	            // إثباتُ الغيابِ بجردٍ قرائيّ ⇒ مسارُ الإنتاجِ (ReconcileOne) يستدعي EndOfLife:
   971	            endOfLife.Invoke(ctx, new object[] { primary });
   972	            object[] ra = new object[] { primary, null, null, null };
   973	            bool stillKnown = (bool)tryObject.Invoke(ctx, ra);
   974	            lines.Add("endoflife: contextRemoved=" + (!stillKnown));
   975	            // إعادةُ استخدامٍ من طرفٍ ثالثٍ بلا تسجيلٍ: المقبضُ يعودُ كائنًا جديدًا (swtpm يعيدُ مقبضَ Primary المحرّر).
   976	            byte[] pr2Cmd = Cmd.CreatePrimary(new byte[0], Templates.StoragePrimary());
   977	            byte[] pr2 = RequireOk(t, pr2Cmd, "reuse");
   978	            uint reused = U32(pr2, 10);
   979	            object[] ua = new object[] { reused, null, null, null };
   980	            bool inherited = (bool)tryObject.Invoke(ctx, ua);
   981	            lines.Add("reuse-unrecorded: handle=" + Hex.U32(reused) + " inheritedContext=" + inherited);
   982	            // ردُّ Create على الأصلِ المعادِ بلا سياقٍ ⇒ رفضٌ مسمّى (CONTEXT_UNKNOWN) لا ادّعاءُ تلفٍ:
   983	            byte[] cc = Cmd.Create(reused, new byte[0], Templates.ServiceSigningKey());
   984	            byte[] cr = RequireOk(t, cc, "create-on-reused");
   985	            string unrecordedVerdict;
   986	            try
   987	            {
   988	                check.Invoke(null, new object[] { Tpm.CC_Create, cc, Params(cr, 0), ctx });
   989	                unrecordedVerdict = "ACCEPT";
   990	            }
   991	            catch (System.Reflection.TargetInvocationException e) { unrecordedVerdict = "REJECT " + e.InnerException.Message; }
   992	            lines.Add("reuse-unrecorded: createCheck=" + unrecordedVerdict);
   993	            // الحالةُ 2: التسجيلُ بعدَ إعادةِ الاستخدامِ ⇒ مقبولٌ بلا وراثةِ الأسماءِ القديمةِ.
   994	            record.Invoke(ctx, new object[] { Tpm.CC_CreatePrimary, pr2Cmd, new uint[] { reused }, Params(pr2, 1) });
   995	            object[] na = new object[] { reused, null, null, null };
   996	            bool reKnown = (bool)tryObject.Invoke(ctx, na);
   997	            byte[] newName = (byte[])na[1]; byte[] newQn = (byte[])na[2];
   998	            lines.Add("reuse-recorded: known=" + reKnown + " newNameLen=" + newName.Length +
   999	                " nameChanged=" + (Hex.Of(newName) != Hex.Of(oldName)) + " qnChanged=" + (Hex.Of(newQn) != Hex.Of(oldQn)));
  1000	            byte[] cc2 = Cmd.Create(reused, new byte[0], Templates.ServiceSigningKey());
  1001	            byte[] cr2 = RequireOk(t, cc2, "create-on-recorded");
  1002	            string recordedVerdict;
  1003	            try
  1004	            {
  1005	                check.Invoke(null, new object[] { Tpm.CC_Create, cc2, Params(cr2, 0), ctx });
  1006	                recordedVerdict = "ACCEPT";
  1007	            }
  1008	            catch (System.Reflection.TargetInvocationException e) { recordedVerdict = "REJECT " + e.InnerException.Message; }
  1009	            lines.Add("reuse-recorded: createCheck=" + recordedVerdict);
  1010	            // تنظيفُ الكائنِ المعادِ استخدامِهِ قبلَ الحالةِ التاليةِ (تحريرٌ ناجحٌ يُسجَّلُ كالإنتاجِ).
  1011	            byte[] reusedFlushCmd = Cmd.FlushContext(reused);
  1012	            RequireOk(t, reusedFlushCmd, "cleanup-reused");
  1013	            record.Invoke(ctx, new object[] { Tpm.CC_FlushContext, reusedFlushCmd, new uint[0], new byte[0] });
  1014	
  1015	            // الحالةُ 3: فقدُ ردِّ التحريرِ والمقبضُ ما زالَ حاضرًا ⇒ السياقُ محفوظٌ.
  1016	            byte[] p2Cmd = Cmd.CreatePrimary(new byte[0], Templates.StoragePrimary());
  1017	            byte[] p2r = RequireOk(t, p2Cmd, "p2");
  1018	            uint p2 = U32(p2r, 10);
  1019	            record.Invoke(ctx, new object[] { Tpm.CC_CreatePrimary, p2Cmd, new uint[] { p2 }, Params(p2r, 1) });
  1020	            // تحريرٌ لمقبضٍ غيرِ موجودٍ ⇒ خطأٌ؛ المقبضُ الحيّ باقٍ والسياقُ لا يُمسّ.
  1021	            byte[] fr = t.Transmit(Cmd.FlushContext(0x81234567));
  1022	            bool flushFailed = U32(fr, 6) != 0;
  1023	            bool p2Present = HandlePresent(t, 0x80000000, p2);
  1024	            object[] pa = new object[] { p2, null, null, null };
  1025	            bool p2Known = (bool)tryObject.Invoke(ctx, pa);
  1026	            lines.Add("still-present: flushFailed=" + flushFailed + " present=" + p2Present + " contextKept=" + p2Known);
  1027	            RequireOk(t, Cmd.FlushContext(p2), "cleanup-p2");
  1028	
  1029	            // الحالةُ 4: استهلاكُ جلسةِ السياسةِ يُقاسُ مباشرةً (سياسةُ المرساةِ بمفتاحِ الخدمةِ).
  1030	            byte[] primCmd = Cmd.CreatePrimary(new byte[0], Templates.StoragePrimary());
  1031	            uint prim = U32(RequireOk(t, primCmd, "prim"), 10);
  1032	            byte[] pcrCmd = Cmd.Create(prim, new byte[0], Templates.ServiceSigningKey());
  1033	            byte[] pcr = RequireOk(t, pcrCmd, "svc-create");
  1034	            Rd k = new Rd(Params(pcr, 0));
  1035	            byte[] priv = k.B2B(); byte[] pub = k.B2B();
  1036	            byte[] ldCmd = Cmd.Load(prim, new byte[0], priv, pub);
  1037	            byte[] ldr = RequireOk(t, ldCmd, "svc-load");
  1038	            uint key = U32(ldr, 10);
  1039	            record.Invoke(ctx, new object[] { Tpm.CC_Create, pcrCmd, new uint[0], Params(pcr, 0) });
  1040	            record.Invoke(ctx, new object[] { Tpm.CC_Load, ldCmd, new uint[] { key }, Params(ldr, 1) });
  1041	            try
  1042	            {
  1043	                byte[] keyName = Policy.Name(pub);
  1044	                byte[] policy = Policy.AnchorIncrementPolicy(keyName);
  1045	                byte[] nvPub = new Wr().U32(0x01807700).U16(Tpm.ALG_SHA256).U32(Tpm.ANCHOR_ATTRIBUTES).B2B(policy).U16(8).Bytes();
  1046	                RequireOk(t, Cmd.NvDefineSpace(new byte[0], new byte[0], nvPub), "anchor-define");
  1047	                try
  1048	                {
  1049	                    byte[] scmd = Cmd.StartAuthSession(Tpm.SE_POLICY, new byte[32]);
  1050	                    byte[] sr = RequireOk(t, scmd, "policy-session");
  1051	                    uint ps = U32(sr, 10);
  1052	                    byte[] nonce = new Rd(Params(sr, 1)).B2B();
  1053	                    record.Invoke(ctx, new object[] { Tpm.CC_StartAuthSession, scmd, new uint[] { ps }, Params(sr, 1) });
  1054	                    object[] sa = new object[] { ps, null };
  1055	                    bool sessionKnown = (bool)trySessionType.Invoke(ctx, sa);
  1056	                    lines.Add("policy-session: known=" + sessionKnown + " type=" + (byte)sa[1]);
  1057	                    byte[] sg = RequireOk(t, Cmd.Sign(key, new byte[0], Policy.AHash(nonce, 0)), "sign-ahash");
  1058	                    Rd sig = new Rd(Params(sg, 0)); sig.U16(); sig.U16(); byte[] r = sig.B2B(); byte[] s = sig.B2B();
  1059	                    RequireOk(t, Cmd.PolicySigned(key, ps, nonce, 0, r, s), "policysigned");
  1060	                    RequireOk(t, Cmd.PolicyCommandCode(ps, Tpm.CC_NV_Increment), "policycommandcode");
  1061	                    // الأمرُ المُفوِّزُ الناجحُ يستهلكُ الجلسةَ (continueSession=0 في PolicyAuth):
  1062	                    RequireOk(t, Cmd.NvIncrement(0x01807700, Cmd.PolicyAuth(ps, new byte[32])), "authorized-increment");
  1063	                    // قياسٌ مباشرٌ فوريّ: الجلسةُ اختفت قبلَ التنظيفِ وقبلَ تحريرِ المفتاحِ.
  1064	                    bool sessionGone = !HandlePresent(t, 0x02000000, ps);
  1065	                    endOfLife.Invoke(ctx, new object[] { ps });
  1066	                    object[] ca = new object[] { ps, null };
  1067	                    bool sessionTypeRemoved = !((bool)trySessionType.Invoke(ctx, ca));
  1068	                    lines.Add("consumed: sessionGoneImmediately=" + sessionGone + " contextRemoved=" + sessionTypeRemoved);
  1069	                }
  1070	                finally { RequireOk(t, Cmd.NvUndefineSpace(new byte[0], 0x01807700), "anchor-undefine"); }
  1071	            }
  1072	            finally
  1073	            {
  1074	                RequireOk(t, Cmd.FlushContext(key), "cleanup-key");
  1075	                RequireOk(t, Cmd.FlushContext(prim), "cleanup-prim");
  1076	            }
  1077	            lines.Add("probe: done");
  1078	            return lines;
  1079	        }
