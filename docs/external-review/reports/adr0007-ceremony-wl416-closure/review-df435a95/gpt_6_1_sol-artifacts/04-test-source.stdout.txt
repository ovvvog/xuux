   891	            if (!guardianOurs) {
   892	              cleanupFailures.push(`guardian identity not provable before signalling (pid ${guardianPid})`);
   893	            } else {
   894	              // الحصادُ الفعليُّ للوالدِ: حدثُ exit يُطلقُهُ libuv بعدَ waitpid — انتظارٌ يُديرُ حلقةَ
   895	              // الأحداثِ (الإشارةُ المتزامنةُ الحاجبةُ تُؤجّلُ الحصادَ فلا يُثبَتُ الزوالُ أبدًا).
   896	              const exited = new Promise((resolve) => {
   897	                parent.once('exit', resolve);
   898	              });
   899	              process.kill(guardianPid, 'SIGKILL');
   900	              await Promise.race([exited, new Promise((r) => setTimeout(r, 5000))]);
   901	            }
   902	          } catch (e) {
   903	            cleanupFailures.push(`guardian cleanup threw: ${e && e.message}`);
   904	          }
   905	          if (!pidProvenDead(guardianPid)) {
   906	            try {
   907	              for (let i = 0; i < 250; i++) {
   908	                if (pidProvenDead(guardianPid)) break;
   909	                await new Promise((r) => setTimeout(r, 20));
   910	              }
   911	            } catch (e) {
   912	              cleanupFailures.push(`guardian wait threw: ${e && e.message}`);
   913	            }
   914	            if (!pidProvenDead(guardianPid))
   915	              cleanupFailures.push(`guardian not provably dead after SIGKILL (pid ${guardianPid})`);
   916	          }
   917	        }
   918	      }
   919	    }
   920	    // (2) الابنُ — مسارٌ مستقلٌّ لا يُعطَّلُ بفشلِ مسارِ الحارسِ (جولةُ 7): حيٌّ ⇒ يُنهى عبرَ
   921	    // القناةِ بهويّتِهِ؛ زومبي ⇒ init يُلقّيهِ بعدَ موتِ والدِهِ.
   922	    if (childPid === null) {
   923	      if (parent !== null)
   924	        cleanupFailures.push('child pid unprovable (no discovery, no valid guardian record)');
   925	    } else if (!pidProvenDead(childPid)) {
   926	      try {
   927	        let st = null;
   928	        try {
   929	          const s = readFileSync(`/proc/${childPid}/stat`, 'utf8');
   930	          const cl = s.lastIndexOf(')');
   931	          if (cl >= 0) st = s.slice(cl + 2).split(' ')[0];
   932	        } catch {
   933	          /* زالت */
   934	        }
   935	        if (st !== 'Z') {
   936	          if (!killSim(childPid, { dir, graceMs: 500 }))
   937	            cleanupFailures.push(`child not stopped via the channel (pid ${childPid})`);
   938	        }
   939	        for (let i = 0; i < 250; i++) {
   940	          if (pidProvenDead(childPid)) break;
   941	          await new Promise((r) => setTimeout(r, 20));
   942	        }
   943	        if (!pidProvenDead(childPid))
   944	          cleanupFailures.push(`child not provably dead (pid ${childPid})`);
   945	      } catch (e) {
   946	        cleanupFailures.push(`child cleanup threw: ${e && e.message}`);
   947	      }
   948	    }
   949	    // (3) بقايا المسارِ: نتائجٌ في الاكتشافِ بعدَ التنظيفِ ⇒ فشلُ تنظيفٍ صريحٌ معلَنٌ يُفشِلُ
   950	    // الاختبارَ — لا مجردَ منعِ الحذفِ (جولةُ 7).
   951	    let remaining = [];
   952	    try {
   953	      remaining = discoverSimPids(dir);
   954	    } catch (e) {
   955	      cleanupFailures.push(`final scan threw: ${e && e.message}`);
   956	    }
   957	    if (remaining.length > 0)
   958	      cleanupFailures.push(`processes still own the state dir after cleanup: ${remaining.join(',')}`);
   959	    // الحذفُ لا يقعُ إلّا ببرهانٍ كاملٍ: لا فشلَ تنظيفٍ ولا عمليّةَ تملكُ المسارَ بعدَ التنظيفِ.
   960	    if (cleanupFailures.length === 0 && remaining.length === 0) {
   961	      rmSync(dir, { recursive: true, force: true });
   962	    }
   963	    // فشلُ التنظيفِ مع فشلِ الجسمِ: يُسجَّلُ صراحةً في الخرجِ (الاختبارُ فاشلٌ أصلًا
   964	    // باستثناءِ الجسمِ فلا يُحجبُ)؛ ومع نجاحِ الجسمِ يُفشِلُ التنظيفُ الاختبارَ صراحةً بعدَ finally.
   965	    if (cleanupFailures.length > 0 && bodyError !== null) {
   966	      console.error(`ZOMBIE cleanup also failed — state dir preserved: ${dir} — ${cleanupFailures.join(' | ')}`);
   967	    }
   968	  }
   969	  if (cleanupFailures.length > 0) {
   970	    throw new Error(
   971	      `ZOMBIE cleanup failed — state dir preserved: ${dir} — ${cleanupFailures.join(' | ')}`,
   972	    );
   973	  }
   974	});
