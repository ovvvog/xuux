   130	    public static class Sha
   131	    {
   132	        public static byte[] H(byte[] data)
   133	        {
   134	            using (SHA256 s = SHA256.Create()) { return s.ComputeHash(data); }
   135	        }
   136	
   137	        public static byte[] Concat(params byte[][] parts)
   138	        {
   139	            int n = 0;
   140	            for (int i = 0; i < parts.Length; i++) n += parts[i].Length;
   141	            byte[] r = new byte[n];
   142	            int o = 0;
   143	            for (int i = 0; i < parts.Length; i++) { Buffer.BlockCopy(parts[i], 0, r, o, parts[i].Length); o += parts[i].Length; }
   144	            return r;
   145	        }
   146	    }

   148	    public sealed class Wr
   149	    {
   150	        private readonly MemoryStream _m = new MemoryStream();
   151	        public Wr U8(byte v) { _m.WriteByte(v); return this; }
   152	        public Wr U16(ushort v) { _m.WriteByte((byte)(v >> 8)); _m.WriteByte((byte)v); return this; }
   153	        public Wr U32(uint v) { U16((ushort)(v >> 16)); U16((ushort)v); return this; }
   154	        public Wr Raw(byte[] b) { _m.Write(b, 0, b.Length); return this; }
   155	        public Wr B2B(byte[] b)
   156	        {
   157	            if (b == null) b = new byte[0];
   158	            if (b.Length > 0xFFFF) throw new ArgumentException("TPM2B too large");
   159	            U16((ushort)b.Length); return Raw(b);
   160	        }
   161	        public byte[] Bytes() { return _m.ToArray(); }
   162	    }

   164	    public sealed class Rd
   165	    {
   166	        private readonly byte[] _b;
   167	        private int _o;
   168	        public Rd(byte[] b) { _b = b; _o = 0; }
   169	        public int Remaining { get { return _b.Length - _o; } }
   170	        private void Need(int n) { if (_o + n > _b.Length) throw new CeremonyFailedException("RESPONSE_TRUNCATED", 0); }
   171	        public byte U8() { Need(1); return _b[_o++]; }
   172	        public ushort U16() { Need(2); ushort v = (ushort)((_b[_o] << 8) | _b[_o + 1]); _o += 2; return v; }
   173	        public uint U32() { Need(4); uint v = ((uint)_b[_o] << 24) | ((uint)_b[_o + 1] << 16) | ((uint)_b[_o + 2] << 8) | _b[_o + 3]; _o += 4; return v; }
   174	        public byte[] Raw(int n) { Need(n); byte[] r = new byte[n]; Buffer.BlockCopy(_b, _o, r, 0, n); _o += n; return r; }
   175	        public byte[] B2B() { ushort n = U16(); return Raw(n); }
   176	    }

   186	        public static byte[] PolicyAuth(uint session, byte[] nonceCaller)
   187	        {
   188	            // continueSession = 0: تُغلَقُ الجلسةُ بعدَ الأمرِ الناجح. HMAC فارغٌ (لا PolicyAuthValue/PolicyPassword).
   189	            return new Wr().U32(session).B2B(nonceCaller).U8(0x00).B2B(new byte[0]).Bytes();

   191	
   192	        public static byte[] Build(uint cc, uint[] handles, byte[][] auths, byte[] parameters)
   193	        {
   194	            Wr body = new Wr();
   195	            for (int i = 0; i < handles.Length; i++) body.U32(handles[i]);
   196	            bool sessions = auths != null && auths.Length > 0;
   197	            if (sessions)
   198	            {
   199	                Wr a = new Wr();
   200	                for (int i = 0; i < auths.Length; i++) a.Raw(auths[i]);
   201	                byte[] ab = a.Bytes();
   202	                body.U32((uint)ab.Length).Raw(ab);
   203	            }
   204	            body.Raw(parameters ?? new byte[0]);
   205	            byte[] b = body.Bytes();
   206	            return new Wr().U16(sessions ? Tpm.ST_SESSIONS : Tpm.ST_NO_SESSIONS).U32((uint)(10 + b.Length)).U32(cc).Raw(b).Bytes();
   207	        }

   356	            return Sha.Concat(new byte[] { 0x00, 0x0B }, Sha.H(publicArea));
   357	        }

   367	    // النقلُ الوحيدُ في هذا الملف: swtpm على loopback. لا TBS ولا جهاز.
   368	    public sealed class SwtpmLoopbackTransport : ICeremonyTransport, IDisposable
   369	    {
   370	        private readonly TcpClient _client;
   371	        private readonly NetworkStream _stream;
   372	
   373	        public SwtpmLoopbackTransport(string host, int port)
   374	        {
   375	            if (host != "127.0.0.1" && host != "::1") throw new CeremonyFailedException("SIMULATOR_TRANSPORT_LOOPBACK_ONLY", 0);
   376	            if (port < 1 || port > 65535) throw new CeremonyFailedException("SIMULATOR_PORT_INVALID", 0);
   377	            _client = new TcpClient(host, port);
   378	            _stream = _client.GetStream();
   379	            _stream.ReadTimeout = 15000;
   380	        }
   381	
   382	        public string Kind { get { return "simulator:swtpm-tcp"; } }
   383	
   384	        public byte[] Transmit(byte[] command)
   385	        {
   386	            _stream.Write(command, 0, command.Length);
   387	            byte[] head = ReadExact(10);
   388	            uint size = new Rd(head).U32Skip2();
   389	            if (size < 10 || size > 65536) throw new CeremonyFailedException("SIMULATOR_RESPONSE_SIZE", 0);
   390	            byte[] rest = ReadExact((int)size - 10);
   391	            return Sha.Concat(head, rest);
   392	        }
   393	
   394	        private byte[] ReadExact(int n)
   395	        {
   396	            byte[] b = new byte[n];
   397	            int got = 0;
   398	            while (got < n)
   399	            {
   400	                int k = _stream.Read(b, got, n - got);
   401	                if (k <= 0) throw new CeremonyFailedException("SIMULATOR_CONNECTION_CLOSED", 0);
   402	                got += k;
   403	            }
   404	            return b;
   405	        }
   406	
   407	        public void Dispose() { _stream.Dispose(); _client.Close(); }
   408	    }

   450	    public sealed class Resp
   451	    {
   452	        public ushort Tag;
   453	        public uint Rc;
   454	        public uint[] Handles = new uint[0];
   455	        public byte[] Params = new byte[0];
   456	
   457	        // تدقيقٌ صارمٌ مربوطٌ بشكلِ الطلبِ (WL-413/SOL-01): وسمُ الردِّ عندَ النجاحِ يطابقُ وسمَ الطلبِ،
   458	        // وردُّ الخطأِ رأسٌ فقط (10 بايتاتٍ) بوسمِ بلا جلساتٍ، ومناطقُ تفويضِ الردِّ بعددِ جلساتِ الطلبِ
   459	        // تُفكُّ وتُستهلَكُ كاملةً — أيُّ تشويهٍ يُفشِلُ مغلقًا برمزٍ مسمّى، لا يُتجاوَزُ إلى الافتراضات.
   460	        public static Resp Parse(byte[] raw, int handleCount, ushort expectedTag, int authCount, bool[] passwordSessions)
   461	        {
   462	            if (raw == null || raw.Length < 10) throw new CeremonyFailedException("RESPONSE_TRUNCATED_HEADER", 0);
   463	            Rd r = new Rd(raw);
   464	            Resp p = new Resp();
   465	            p.Tag = r.U16();
   466	            uint size = r.U32();
   467	            p.Rc = r.U32();
   468	            if (size != raw.Length) throw new ResponseMalformedException("RESPONSE_SIZE_MISMATCH", p.Rc);
   469	            if (p.Rc != 0)
   470	            {
   471	                // ردُّ الخطأِ لا يحملُ مقابضَ ولا معاملاتٍ ولا تفويضاتٍ (Part 3 §6.1): رأسٌ فقط.
   472	                if (p.Tag != Tpm.ST_NO_SESSIONS) throw new ResponseMalformedException("RESPONSE_ERROR_TAG_MALFORMED", p.Rc);
   473	                if (raw.Length != 10) throw new ResponseMalformedException("RESPONSE_ERROR_BODY_MALFORMED", p.Rc);
   474	                return p;
   475	            }
   476	            if (p.Tag != expectedTag)
   477	                throw new ResponseMalformedException("RESPONSE_TAG_MISMATCH: expected=0x" + expectedTag.ToString("x4") + " got=0x" + p.Tag.ToString("x4"), 0);
   478	            p.Handles = new uint[handleCount];
   479	            for (int i = 0; i < handleCount; i++) p.Handles[i] = r.U32();
   480	            if (p.Tag == Tpm.ST_SESSIONS)
   481	            {
   482	                if (authCount < 1) throw new CeremonyFailedException("RESPONSE_AUTH_COUNT_MALFORMED", 0);
   483	                uint ps = r.U32();
   484	                // حدٌّ صريحٌ قبلَ التخصيصِ (WL-413 الدفعة الثالثة / SOL413-07): إعلانُ حجمٍ يفوقُ ما تبقّى
   485	                // من الردِّ فشلٌ مسمّى لا تخصيصًا ضخمًا ولا اقتطاعًا صامتًا.
   486	                if (ps > (uint)r.Remaining)
   487	                    throw new ResponseMalformedException("RESPONSE_PARAMETER_SIZE_INVALID: " + ps + " > remaining " + r.Remaining, 0);
   488	                p.Params = r.Raw((int)ps);
   489	                try
   490	                {
   491	                    // منطقةُ تفويضِ الردِّ: TPMS_AUTH_RESPONSE لكلِّ جلسةٍ أُرسِلَت في الطلبِ (Part 2 §10.13.3).
   492	                    // تدقيقُ المحتوى (WL-413 الدفعة الثالثة / SOL413-03): البروفةُ لا تُرسِلُ إلا نوعيْ تفويضٍ —
   493	                    // كلمةَ سرٍّ (RS_PW) وجلسةَ سياسةٍ — فكلُّ ردِّ تفويضٍ يجبُ أن يطابقَ أحدَ الملفيْنِ كاملًا
   494	                    // وإلّا فشلٌ مغلقٌ مسمّى لا يُمرَّرُ:
   495	                    //   RS_PW  : nonceTPM فارغٌ (حجمُ 0)، sessionAttributes = continueSession وحده (0x01)، hmac فارغٌ.
   496	                    //   policy : nonceTPM من 16..64 بايتًا، hmac فارغٌ (لا sessionKey لهذهِ الجلساتِ)،
   497	                    //            sessionAttributes بلا بتّاتٍ محفوظةٍ (أعلىُ 4 بتّاتٍ صفرٌ دائمًا).
   498	                    for (int i = 0; i < authCount; i++)
   499	                    {
   500	                        byte[] nonce = r.B2B();
   501	                        byte attrs = r.U8();
   502	                        byte[] hmac = r.B2B();
   503	                        bool passwordProfile = nonce.Length == 0 && hmac.Length == 0 && attrs == 0x01;
   504	                        bool policyProfile = nonce.Length == 32 && hmac.Length == 0 && attrs == 0x00; // قياسُ swtpm: بلا continueSession
   505	                        bool expectPassword = passwordSessions != null && i < passwordSessions.Length && passwordSessions[i];
   506	                        if (expectPassword ? !passwordProfile : !policyProfile)
   507	                            throw new ResponseMalformedException("RESPONSE_AUTH_CONTENT_MALFORMED: auth#" + i + " nonceLen=" + nonce.Length + " attrs=0x" + attrs.ToString("x2") + " hmacLen=" + hmac.Length + " expected=" + (expectPassword ? "password" : "policy"), 0);
   508	                    }
   509	                }
   510	                catch (ResponseMalformedException) { throw; }
   511	                catch (Exception)
   512	                {
   513	                    throw new CeremonyFailedException("RESPONSE_AUTH_AREA_MALFORMED", 0);
   514	                }
   515	                if (r.Remaining != 0) throw new CeremonyFailedException("RESPONSE_TRAILING_BYTES", 0);
   516	            }
   517	            else
   518	            {
   519	                p.Params = r.Raw(r.Remaining);
   520	            }
   521	            return p;
   522	        }
   523	    }

   525	    // ── مخططُ بنيةِ معاملاتِ الردِّ الناجحِ لكلِّ أمرٍ (WL-413 الدفعة الثانية / R2-03): يُستهلَكُ كلُّ
   526	    // حقلٍ ويُتحقَّقُ من الأحجامِ الداخليّةِ ومن انعدامِ البايتاتِ الزائدةِ. أمرٌ بلا مخطّطٍ يُرفَضُ — لا
   527	    // يُفسَّرُ ردٌّ لا يُعرَفُ شكلُه. البنى من TPM 2.0 Part 3 (ردودُ الأوامر) وPart 2 (الأنواع). ──
   528	    // سياقُ الردودِ المُتحقَّقِ منها (WL-414 دورةُ المعالجةِ / FUP-SOL-01/02/05): يسجّلُ — بعدَ نجاحِ فحصِ
   529	    // المخطّطِ فقط — اسمَ كلِّ كائنٍ أنشأتْه البروفةُ أو حمّلتْه، وهرميّتَه، واسمَه المؤهَّلَ، ونوعَ كلِّ جلسةٍ
   530	    // بدأتْها. فحوصُ الردودِ التي تعتمدُ على حالةٍ سابقةٍ (اسمُ الأصلِ في بياناتِ الإنشاءِ، وتذكرةُ PolicySigned
   531	    // الحقيقيّةُ عندَ expiration سالبٍ) تُربَطُ به؛ مقبضٌ غيرُ مسجّلٍ ⇒ فشلٌ مغلقٌ مسمّى لا قبولٌ بنيويٌّ.
   532	    public sealed class ResponseContext
   533	    {
   534	        private readonly Dictionary<uint, byte[]> _names = new Dictionary<uint, byte[]>();
   535	        private readonly Dictionary<uint, byte[]> _qualified = new Dictionary<uint, byte[]>();
   536	        private readonly Dictionary<uint, uint> _hierarchy = new Dictionary<uint, uint>();
   537	        private readonly Dictionary<uint, byte> _sessionType = new Dictionary<uint, byte>();
   538	
   539	        public bool TryObject(uint h, out byte[] name, out byte[] qualifiedName, out uint hierarchy)
   540	        {
   541	            name = null; qualifiedName = null; hierarchy = 0;
   542	            if (!_names.ContainsKey(h)) return false;
   543	            name = _names[h]; qualifiedName = _qualified[h]; hierarchy = _hierarchy[h];
   544	            return true;
   545	        }
   546	
   547	        public bool TrySessionType(uint h, out byte type)
   548	        {
   549	            type = 0;
   550	            if (!_sessionType.ContainsKey(h)) return false;
   551	            type = _sessionType[h];
   552	            return true;
   553	        }
   554	
   555	        // هل بقيَ للعمليّةِ سياقٌ مسجَّلٌ؟ — شاهدُ الحالةِ الفعليّةِ لا مجاوَرةِ النداءِ: القيمةُ تُشتَقُّ من
   556	        // الخرائطِ نفسِها بعدَ استدعاءاتِ الإزالةِ، فحذفُ EndOfLife مع إبقاءِ الحقيقةِ يُسقطُ الشاهدَ (جولةُ 4).
   557	        public bool Knows(uint h)
   558	        {
   559	            return _names.ContainsKey(h) || _sessionType.ContainsKey(h);
   560	        }
   561	
   562	        // الاسمُ المؤهَّلُ (Part 1 §16): QN(B) = nameAlg || H(QN(أصلِ B) || Name(B))، وQN للهرميّةِ = مقبضُها (4 بايتات).
   563	        // قياسُ swtpm 2026-10-18: parentQualifiedName في ردِّ Create = 000B || SHA256(40000001 || Name(primary)).
   564	        public static byte[] Qualified(byte[] parentQn, byte[] name)
   565	        {
   566	            byte[] cat = new byte[parentQn.Length + name.Length];
   567	            Array.Copy(parentQn, 0, cat, 0, parentQn.Length);
   568	            Array.Copy(name, 0, cat, parentQn.Length, name.Length);
   569	            byte[] d = Sha.H(cat);
   570	            byte[] q = new byte[2 + d.Length];
   571	            q[0] = 0x00; q[1] = 0x0B;
   572	            Array.Copy(d, 0, q, 2, d.Length);
   573	            return q;
   574	        }
   575	
   576	        private static uint U32At(byte[] b, int o) { return ((uint)b[o] << 24) | ((uint)b[o + 1] << 16) | ((uint)b[o + 2] << 8) | b[o + 3]; }
   577	
   578	        public static byte[] HierarchyName(uint h) { return new byte[] { (byte)(h >> 24), (byte)(h >> 16), (byte)(h >> 8), (byte)h }; }
   579	
   580	        // يُستدعى بعدَ ResponseSchema.Check الناجحِ فقط (ردُّ رمزِه صفرٌ ومخطّطُه صالح).
   581	        public void Record(uint cc, byte[] cmd, uint[] handles, byte[] prm)
   582	        {
   583	            switch (cc)
   584	            {
   585	                case Tpm.CC_CreatePrimary:
   586	                {
   587	                    uint hier = U32At(cmd, 10);
   588	                    Rd r = new Rd(prm); r.B2B(); r.B2B(); r.B2B(); r.U16(); r.U32(); r.B2B();
   589	                    byte[] nm = r.B2B();
   590	                    Put(handles[0], nm, Qualified(HierarchyName(hier), nm), hier);
   591	                    return;
   592	                }
   593	                case Tpm.CC_Load:
   594	                {
   595	                    uint parent = U32At(cmd, 10);
   596	                    byte[] pn, pq; uint ph;
   597	                    if (!TryObject(parent, out pn, out pq, out ph)) return; // أصلٌ غيرُ معروفٍ ⇒ لا يُسجَّلُ الابنُ فتُرفضُ فحوصُه لاحقًا
   598	                    byte[] nm = new Rd(prm).B2B();
   599	                    Put(handles[0], nm, Qualified(pq, nm), ph);
   600	                    return;
   601	                }
   602	                case Tpm.CC_StartAuthSession:
   603	                {
   604	                    Rd c = new Rd(cmd); c.Raw(10 + 8); c.B2B(); c.B2B();
   605	                    // مقبضٌ كان كائنًا ثمّ عادَ جلسةً: سياقُ الكائنِ لا يُورَّثُ للجلسةِ (WL-416) — دورةُ حياةِ
   606	                    // السياقِ تتبعُ دورةَ حياةِ المقبضِ، والعكسُ موجودٌ سلفًا في Put (إزالةُ نوعِ الجلسةِ).
   607	                    _names.Remove(handles[0]); _qualified.Remove(handles[0]); _hierarchy.Remove(handles[0]);
   608	                    _sessionType[handles[0]] = c.U8();
   609	                    return;
   610	                }
   611	                case Tpm.CC_FlushContext:
   612	                {
   613	                    uint h = U32At(cmd, 10);
   614	                    _names.Remove(h); _qualified.Remove(h); _hierarchy.Remove(h); _sessionType.Remove(h);
   615	                    return;
   616	                }
   617	            }
   618	        }
   619	
   620	        private void Put(uint h, byte[] name, byte[] qn, uint hier)
   621	        {
   622	            _names[h] = name; _qualified[h] = qn; _hierarchy[h] = hier; _sessionType.Remove(h);
   623	        }
   624	
   625	        // ── إزالةُ السياقِ عندَ إثباتِ انتهاءِ العمرِ (WL-416) ──
   626	        // تُربطُ بالحدثِ الذي يُثبتُ زوالَ المقبضِ أو الجلسةِ فعلًا، لا بمجردِ نتيجةٍ مجهولةٍ:
   627	        //   1. نجاحُ FlushContext المؤكَّد (عبرَ Record عندَ رمزِ صفرٍ).
   628	        //   2. جردٌ قرائيٌّ يُثبتُ غيابَ المقبضِ بعدَ فقدِ ردِّ التحريرِ (مسارُ المصالحةِ).
   629	        //   3. نجاحُ العمليةِ التي تستهلكُ الجلسةَ (جلسةُ السياسةِ تُستهلَكُ بأمرِ التفويضِ الناجحِ —
   630	        //      سلوكُ المحاكي المُثبَتُ: الجردُ النهائيّ لا يُبقي جلساتٍ مستعملةً).
   631	        // المقبضُ الحيّ أو الجردُ غيرُ المحسومِ لا يُزيلُ السياقَ — الفشلُ المغلقُ باقٍ، والسياقُ المفقودُ لا
   632	        // يتحوّلُ قبولًا افتراضيًّا: فحوصُ الكائناتِ والجلساتِ المجهولةِ تظلُّ مسمّاةً (CONTEXT_UNKNOWN).
   633	        public void EndOfLife(uint h)
   634	        {
   635	            _names.Remove(h); _qualified.Remove(h); _hierarchy.Remove(h); _sessionType.Remove(h);
   636	        }
   637	    }

   641	        private static void End(Rd r, string what)
   642	        {
   643	            if (r.Remaining != 0) throw new CeremonyFailedException("RESPONSE_PARAMS_TRAILING_BYTES: " + what + " extra=" + r.Remaining, 0);
   644	        }
   645	
   646	        private static void Name(byte[] n, string what)
   647	        {
   648	            // TPM2B_NAME لكائنٍ/فهرسٍ بخوارزميّةِ SHA-256: nameAlg(2) || digest(32).
   649	            if (n.Length != 34 || n[0] != 0x00 || n[1] != 0x0B) throw new CeremonyFailedException("RESPONSE_NAME_MALFORMED: " + what + " len=" + n.Length, 0);
   650	        }
   651	
   652	        // عددُ مقابضِ الطلبِ لكلِّ أمرٍ تستعملُه البروفةُ (Part 3) — لاستخراجِ منطقةِ معاملاتِ الطلبِ.
   653	        // نوعُ كلِّ جلسةٍ في طلبٍ ما: RS_PW (كلمةُ سرٍّ) أم مقبضُ جلسةٍ (سياسة/تجريبية) — من بايتاتِ الطلبِ
   654	        // نفسِه. يُربَطُ ملفُّ تفويضِ الردِّ بنوعِ الجلسةِ المُرسَلةِ (R3SOL-02/قياسُ المراجعينَ الثلاثةِ):
   655	        // ملفُّ السياسةِ في ردِّ أمرٍ بكلمةِ سرٍّ — والعكسُ — كان يُقبلُ لأنّ الفاحصَ يقبلُ أحدَ الملفينِ
   656	        // على أيِّ منطقةِ تفويضٍ. attrs=0x01 (continueSession وحده) للملفينِ، وnonceTPM للسياسةِ = 32
   657	        // (حجمُ nonceCaller في هذهِ البروفةِ؛ nonceTPM يُحفظُ من أولِّ nonceCaller — Part 1 §19.6.3.2).

   716	        // تذكرةُ إنشاءٍ (TPMT_TK_CREATION): الوسمُ 0x8021 (TPM_ST_CREATED) والهرميّةُ مقبضُ ثابتٍ صالحٌ

   722	            // NULL 0x40000007، التأييدُ 0x4000000B، المنصّةُ 0x4000000C) — نطاقٌ عدديٌّ متصلٌ يقبلُ

   724	            if (hierarchy != 0x40000001u && hierarchy != 0x40000007u && hierarchy != 0x4000000Bu && hierarchy != 0x4000000Cu)

   733	        // تذكرةُ تفويضٍ (TPMT_TK_AUTH) في ردِّ PolicySigned (WL-414 FUP-SOL-01/05 + ربطُ طولِ البصمةِ): الوسمُ
   734	        // TPM_ST_AUTH_SIGNED = 0x8025 حصرًا (Part 2 جدولُ 19؛ 0x8023 هو AUTH_SECRET لـPolicySecret — كان يُقبلُ
   735	        // خطأً بتعليقٍ عكسَ الوسمينِ). شكلُ التذكرةِ يُستنتَجُ من الطلبِ لا من الردِّ (Part 3 §23.3 + قياسُ swtpm):
   736	        //   expiration ≥ 0، أو جلسةُ تجربةٍ ⇒ تذكرةُ NULL: timeout فارغٌ، hierarchy = RH_NULL، بصمةٌ فارغةٌ.
   737	        //   expiration < 0 في جلسةِ سياسةٍ ⇒ تذكرةٌ حقيقيّةٌ: timeout بـ8 بايتاتٍ (UINT64)، hierarchy = هرميّةُ
   738	        //   authObject المسجّلةُ، بصمةٌ بـTicketDigestLen.
   739	        // سابقًا: كانت أيُّ هرميّةٍ من الأربعِ وأيُّ بصمةٍ ≤64 تُقبلُ عندَ الصفرِ، وكان الردُّ الصحيحُ عندَ
   740	        // expiration موجبٍ (timeout فارغٌ) يُرفضُ.
   741	        private static void PolicySignedTicket(byte[] cmd, byte[] to, ushort tag, uint hierarchy, byte[] digest, ResponseContext ctx)
   742	        {
   743	            if (tag != 0x8025)
   744	                throw new ResponseMalformedException("RESPONSE_TICKET_TAG_MALFORMED: PolicySigned 0x" + tag.ToString("x4"), 0);
   745	            Rd cp = new Rd(CmdParams(cmd)); cp.B2B(); cp.B2B(); cp.B2B(); int expiration = (int)cp.U32();
   746	            bool realTicket = false;
   747	            uint expectedHierarchy = Tpm.RH_NULL;
   748	            if (expiration < 0)
   749	            {
   750	                uint authObject = ((uint)cmd[10] << 24) | ((uint)cmd[11] << 16) | ((uint)cmd[12] << 8) | cmd[13];
   751	                uint session = ((uint)cmd[14] << 24) | ((uint)cmd[15] << 16) | ((uint)cmd[16] << 8) | cmd[17];
   752	                byte type; byte[] n, q; uint h;
   753	                if (ctx == null || !ctx.TrySessionType(session, out type))
   754	                    throw new ResponseMalformedException("RESPONSE_POLICY_CONTEXT_UNKNOWN: session " + Hex.U32(session) + " at expiration=" + expiration, 0);
   755	                if (type == Tpm.SE_POLICY)
   756	                {
   757	                    if (!ctx.TryObject(authObject, out n, out q, out h))
   758	                        throw new ResponseMalformedException("RESPONSE_POLICY_CONTEXT_UNKNOWN: authObject " + Hex.U32(authObject), 0);
   759	                    realTicket = true;
   760	                    expectedHierarchy = h;
   761	                }
   762	            }
   763	            int expectedTimeout = realTicket ? 8 : 0;
   764	            int expectedDigest = realTicket ? TicketDigestLen : 0;
   765	            if (to.Length != expectedTimeout)
   766	                throw new ResponseMalformedException("RESPONSE_POLICY_TIMEOUT_MALFORMED: len=" + to.Length + " expected=" + expectedTimeout + " at expiration=" + expiration, 0);
   767	            if (hierarchy != expectedHierarchy)
   768	                throw new ResponseMalformedException("RESPONSE_TICKET_HIERARCHY_MALFORMED: PolicySigned " + Hex.U32(hierarchy) + " expected=" + Hex.U32(expectedHierarchy), 0);
   769	            if (digest.Length != expectedDigest)
   770	                throw new ResponseMalformedException("RESPONSE_TICKET_DIGEST_MALFORMED: PolicySigned len=" + digest.Length + " expected=" + expectedDigest, 0);
   771	        }

   774	        // المُرسَلَ بايتًا ببايتٍ، وunique نقطةُ P-256 كاملةٌ: X وY بـ32 بايتًا لكلٍّ منهما.

   788	        // TPMS_CREATION_DATA (SOL413-04) كما يُعلّمُها swtpm/libtpms (قياسُ 2026-10-18): تحديدُ PCR بعددٍ
   789	        // ومداخلَ محدودةٍ، ثمّ بصمةٌ بحجمِ 32، ثمّ locality وparentNameAlg ثمّ حقولُ B2B محدودةٌ تُستهلَكُ كاملةً.
   790	        private static void CreationData(byte[] cd, string what, byte[] cmd, ResponseContext ctx)
   791	        {
   792	            try
   793	            {
   794	                if (cd.Length < 13 || cd.Length > 512)
   795	                    throw new ResponseMalformedException("RESPONSE_CREATION_DATA_MALFORMED: " + what + " len=" + cd.Length, 0);
   796	                Rd q = new Rd(cd);
   797	                uint count = q.U32();
   798	                if (count > 4) throw new ResponseMalformedException("RESPONSE_CREATION_DATA_MALFORMED: " + what + " pcrSelect count=" + count, 0);
   799	                for (uint i = 0; i < count; i++)
   800	                {
   801	                    q.U16();
   802	                    byte sz = q.U8();
   803	                    if (sz > 16) throw new ResponseMalformedException("RESPONSE_CREATION_DATA_MALFORMED: " + what + " pcrSelect size=" + sz, 0);
   804	                    q.Raw(sz);
   805	                }
   806	                // بصمةُ PCR: 32 (قياسُ swtpm عندَ قائمةٍ فارغةٍ) أو 0 عندَ قائمةٍ فارغةٍ (جملةُ الجدولِ
   807	                // 222 في Part 2: pcrDigest.size صفرٌ عندَ قائمةٍ فارغةٍ) — كلا الترميزينِ سائغٌ (R3SOL-05).
   808	                ushort dlen = q.U16();
   809	                if (dlen != 32 && !(count == 0 && dlen == 0))
   810	                    throw new ResponseMalformedException("RESPONSE_CREATION_DATA_MALFORMED: " + what + " pcrDigest len=" + dlen, 0);
   811	                q.Raw(dlen);
   812	                q.U8();  // locality
   813	                // خوارزميّةُ اسمِ الأصلِ: ALG_NULL لإنشاءِ Primary بلا أصلٍ (قياسُ swtpm)، أو SHA-256
   814	                // لإنشاءِ مفتاحٍ ابنٍ — ما عدا هذينِ فشلٌ مسمّى (قياسُ المراجعينَ: 0xffff كان يُقبلُ).
   815	                ushort pna = q.U16();
   816	                if (pna != Tpm.ALG_NULL && pna != Tpm.ALG_SHA256)
   817	                    throw new ResponseMalformedException("RESPONSE_CREATION_DATA_MALFORMED: " + what + " parentNameAlg=0x" + pna.ToString("x4"), 0);
   818	                // ثلاثةُ حقولِ B2B ذيليّةٍ حصرًا بترتيبِ Part 2 جدولِ 212: parentName ثمّ parentQualifiedName
   819	                // ثمّ outsideInfo — حقولٌ ناقصةٌ أو رابعٌ زائدٌ كان يُقبلُ (قياسُ المراجعينَ).
   820	                byte[][] tail = new byte[3][];
   821	                int fields = 0;
   822	                while (q.Remaining >= 2 && fields < 3) { tail[fields] = q.B2B(); fields++; }
   823	                if (fields != 3 || q.Remaining != 0)
   824	                    throw new ResponseMalformedException("RESPONSE_CREATION_DATA_MALFORMED: " + what + " tailFields=" + fields + " tail=" + q.Remaining, 0);
   825	                // ربطُ الحقولِ الذيليّةِ بالطلبِ وبالأصلِ المعروفِ (WL-414 FUP-SOL-02): كانت تُستهلَكُ بنيويًّا
   826	                // فقط، فقُبلَ outsideInfo مختلفٌ واسمُ أصلٍ مقلوبٌ (قياسُ المراجعِ على 3e67d362 وa3272058).
   827	                Rd cp = new Rd(CmdParams(cmd)); cp.B2B(); cp.B2B(); byte[] askedOutside = cp.B2B();
   828	                if (Hex.Of(tail[2]) != Hex.Of(askedOutside))
   829	                    throw new ResponseMalformedException("RESPONSE_CREATION_DATA_MALFORMED: " + what + " outsideInfo differs from request", 0);
   830	                uint parent = ((uint)cmd[10] << 24) | ((uint)cmd[11] << 16) | ((uint)cmd[12] << 8) | cmd[13];
   831	                byte[] expName, expQn; ushort expAlg;
   832	                if (CommandCodeOfCmd(cmd) == Tpm.CC_CreatePrimary)
   833	                {
   834	                    // Primary: لا أصلَ كائنيًّا ⇒ parentNameAlg = ALG_NULL، والاسمُ والاسمُ المؤهَّلُ = مقبضُ الهرميّةِ (قياسُ swtpm).
   835	                    expAlg = Tpm.ALG_NULL;
   836	                    expName = ResponseContext.HierarchyName(parent);
   837	                    expQn = expName;
   838	                }
   839	                else
   840	                {
   841	                    uint ph;
   842	                    if (ctx == null || !ctx.TryObject(parent, out expName, out expQn, out ph))
   843	                        throw new ResponseMalformedException("RESPONSE_CREATION_PARENT_UNKNOWN: " + what + " parent " + Hex.U32(parent), 0);
   844	                    expAlg = Tpm.ALG_SHA256;
   845	                }
   846	                if (pna != expAlg)
   847	                    throw new ResponseMalformedException("RESPONSE_CREATION_DATA_MALFORMED: " + what + " parentNameAlg=0x" + pna.ToString("x4") + " expected=0x" + expAlg.ToString("x4"), 0);
   848	                if (Hex.Of(tail[0]) != Hex.Of(expName))
   849	                    throw new ResponseMalformedException("RESPONSE_CREATION_DATA_MALFORMED: " + what + " parentName differs from known parent", 0);
   850	                if (Hex.Of(tail[1]) != Hex.Of(expQn))
   851	                    throw new ResponseMalformedException("RESPONSE_CREATION_DATA_MALFORMED: " + what + " parentQualifiedName differs from known parent", 0);
   852	            }
   853	            catch (ResponseMalformedException) { throw; }
   854	            catch (Exception)
   855	            {
   856	                throw new ResponseMalformedException("RESPONSE_CREATION_DATA_MALFORMED: " + what, 0);
   857	            }
   858	        }

   947	                    if (ch.Length != 32) throw new ResponseMalformedException("RESPONSE_CREATION_HASH_MALFORMED: len=" + ch.Length, 0);
   948	                    // بصمةُ تذكرةِ الإنشاءِ وفحصُ اتساقِ creationHash مع بياناتِ الإنشاءِ (R3SOL-03).
   949	                    if (tdg.Length != TicketDigestLen) throw new ResponseMalformedException("RESPONSE_TICKET_DIGEST_MALFORMED: CreatePrimary len=" + tdg.Length, 0);
   950	                    if (Hex.Of(Sha.H(cd)) != Hex.Of(ch))
   951	                        throw new ResponseMalformedException("RESPONSE_CREATION_HASH_MISMATCH: CreatePrimary", 0);
   952	                    CreationTicket(ttag, h, "CreatePrimary");
   953	                    Name(nm, "CreatePrimary");
   954	                    if (Hex.Of(nm) != Hex.Of(Policy.Name(pub)))
   955	                        throw new ResponseMalformedException("RESPONSE_NAME_HASH_MISMATCH: CreatePrimary", 0);
   956	                    return;
   957	                }
   958	                case Tpm.CC_Sign:
   959	                {
   960	                    ushort alg = r.U16();
   961	                    if (alg != Tpm.ALG_ECDSA) throw new CeremonyFailedException("RESPONSE_SIGNATURE_ALG_UNEXPECTED: 0x" + alg.ToString("x4"), 0);
   962	                    ushort hash = r.U16();

   966	                        throw new ResponseMalformedException("RESPONSE_SIGNATURE_COMPONENT_MALFORMED: r=" + rr.Length + " s=" + ss.Length, 0);
   967	                    return;
   968	                }
   969	                case Tpm.CC_PolicySigned:
   970	                {
   971	                    byte[] to = r.B2B();
   972	                    ushort ttag = r.U16(); uint h = r.U32(); byte[] td = r.B2B();
   973	                    End(r, "PolicySigned");
   974	                    PolicySignedTicket(cmd, to, ttag, h, td, ctx);
   975	                    return;
   976	                }
   977	                case Tpm.CC_GetCapability:

   979	                    Rd c = new Rd(cmd); c.Raw(10); uint askedCap = c.U32(); uint askedProp = c.U32(); uint askedCount = c.U32();
   980	                    byte more = r.U8();
   981	                    if (more > 1) throw new CeremonyFailedException("RESPONSE_CAPABILITY_MORE_MALFORMED", 0);
   982	                    uint cap = r.U32();
   983	                    if (cap != askedCap) throw new CeremonyFailedException("RESPONSE_CAPABILITY_MISMATCH: asked=" + Hex.U32(askedCap) + " got=" + Hex.U32(cap), 0);
   984	                    uint n = r.U32();
   985	                    if (n > 256) throw new CeremonyFailedException("RESPONSE_CAPABILITY_COUNT_IMPLAUSIBLE: " + n, 0);
   986	                    // عددٌ يفوقُ المطلوبَ في الطلبِ فشلٌ مسمّى (R3SOL-01/فحصُ الحدودِ المباشر): الردُّ
   987	                    // يعيدُ أكثرَ ممّا طُلب (65 لطلبِ 64) — الطلبُ هو المرجعُ (Part 3 §30.2).
   988	                    if (n > askedCount) throw new ResponseMalformedException("RESPONSE_CAPABILITY_COUNT_MALFORMED: " + n + " > requested " + askedCount, 0);
   989	                    // تقدّمٌ مزعومٌ بلا أثرٍ (SOL413-02): more=1 مع count=0 فشلٌ مسمّى لا نهايةٌ صامتةٌ للجردِ.
   990	                    if (more == 1 && n == 0) throw new ResponseMalformedException("RESPONSE_CAPABILITY_NO_PROGRESS: more=1 count=0", 0);
   991	                    int width = cap == 1 ? 4 : cap == 6 ? 8 : -1;
   992	                    if (width < 0) throw new CeremonyFailedException("RESPONSE_CAPABILITY_UNSUPPORTED: " + Hex.U32(cap), 0);
   993	                    if (cap == 1)
   994	                    {
   995	                        // فئةُ المقابضِ المعادةِ تطابقُ فئةَ property المطلوبِ (SOL413-02): مقبضٌ من فئةٍ أخرى
   996	                        // في جردِ الجلساتِ/العابرةِ/NV فشلٌ مسمّى — لا إيقافٌ صامتٌ يجعلُ الجردَ ناقصًا.
   997	                        // صعودٌ صارمٌ من قيمةِ الطلبِ (R3SOL-01/فحصُ الحدودِ): المقابضُ تعودُ تصاعديًا من
   998	                        // property فلا مقبضَ دونَه ولا ترتيبًا تنازليًا ولا تكرارًا (Part 3 §30.2).
   999	                        uint prev = 0;
  1000	                        for (uint i = 0; i < n; i++)
  1001	                        {
  1002	                            uint hd = r.U32();
  1003	                            byte cls = (byte)(hd >> 24);
  1004	                            byte want = (byte)(askedProp >> 24);
  1005	                            bool ok = (want == 0x02 || want == 0x03) ? (cls == 0x02 || cls == 0x03) : cls == want;
  1006	                            if (!ok) throw new ResponseMalformedException("RESPONSE_CAPABILITY_HANDLE_CLASS_MALFORMED: " + Hex.U32(hd) + " asked=" + Hex.U32(askedProp), 0);
  1007	                            if (hd < askedProp || (i > 0 && hd <= prev))
  1008	                                throw new ResponseMalformedException("RESPONSE_CAPABILITY_HANDLE_ORDER_MALFORMED: " + Hex.U32(hd) + " after " + Hex.U32(prev) + " from " + Hex.U32(askedProp), 0);
  1009	                            prev = hd;
  1010	                        }
  1011	                    }
  1012	                    else
  1013	                    {
  1014	                        // خاصيّةٌ تعودُ دونَ قيمةِ الطلبِ فشلٌ مسمّى (فحصُ الحدودِ المباشر): الاستئنافُ
  1015	                        // يبدأُ عندَ property أو أعلى (Part 3 §30.2).
  1016	                        for (uint i = 0; i < n; i++)
  1017	                        {
  1018	                            uint prop = r.U32(); r.U32();
  1019	                            if (prop < askedProp) throw new ResponseMalformedException("RESPONSE_CAPABILITY_PROPERTY_MALFORMED: " + Hex.U32(prop) + " < " + Hex.U32(askedProp), 0);
  1020	                        }
  1021	                    }
  1022	                    End(r, "GetCapability"); return;
  1023	                }
  1024	                default:
  1025	                    throw new CeremonyFailedException("RESPONSE_SCHEMA_UNKNOWN_COMMAND: " + Hex.U32(cc), 0);
  1026	            }
  1027	        }

  1206	        private Resp Send(string name, byte[] cmd, int respHandles, string expectation, int authCount)
  1207	        {
  1208	            if (_nvGuardArmed && NvGuardedAtSend(CommandCodeOf(cmd))) AssertNvRangeOwnership(name);
  1209	            Step s = new Step();
  1210	            s.Stage = _cleanup ? "cleanup" : _stage;
  1211	            s.Name = name;
  1212	            s.CommandCode = CommandCodeOf(cmd);
  1213	            s.CommandSha256 = Hex.Of(Sha.H(cmd));
  1214	            s.Expectation = expectation;
  1215	            _r.Steps.Add(s);
  1216	            Resp p;
  1217	            try
  1218	            {
  1219	                byte[] raw = _t.Transmit(cmd);
  1220	                p = Resp.Parse(raw, respHandles, TagOf(cmd), authCount, ResponseSchema.ExpectedPasswordSessions(cmd, authCount));
  1221	                if (p.Rc == 0)
  1222	                {
  1223	                    ResponseSchema.Check(CommandCodeOf(cmd), cmd, p.Params, _ctx);
  1224	                    _ctx.Record(CommandCodeOf(cmd), cmd, p.Handles, p.Params);
  1225	                }
  1226	                // فحصُ تبنّي المقبضِ داخلَ الإرسالِ نفسه (إصلاحٌ R3): مقبضٌ من خطِّ الأساسِ في ردِّ إنشاءٍ يُرمى هنا — فيُغلّفُ الفشلُ للأمرِ
  1227	                // المُغيّرِ بنتيجةٍ مجهولةٍ ويُجرَدُ كسائرِهَا (SOL413-01)، لا فشلٌ مجردًا يفقدُ التسميةَ.
  1228	                for (int i = 0; i < p.Handles.Length; i++) AdoptHandle(p.Handles[i], NameSuffix(name));
  1229	            }
  1230	            catch (CeremonyUnknownOutcomeException) { throw; }
  1231	            catch (Exception e)
  1232	            {
  1233	                // الردُّ لم يُستلَمْ بعدَ إرسالِ أمرٍ مُغيِّرٍ: الأثرُ مجهولٌ — يُفشِلُ مغلقًا ويُجردُ لاحقًا.
  1234	                if (StateChanging.Contains(CommandCodeOf(cmd)))
  1235	                {
  1236	                    s.Outcome = "unknown";
  1237	                    throw new CeremonyUnknownOutcomeException(name, e);
  1238	                }
  1239	                throw;
  1240	            }
  1241	            s.ResponseCode = p.Rc;
  1242	            // رفضُ الطبقةِ (TBS) أو أيُّ رمزِ Windows ⇒ توقّفٌ مغلقٌ: لا إعادة، ولا بديل، ولا تخفيفُ سمات.
  1243	            if ((p.Rc & 0xFFFF0000u) == 0x80280000u)
  1244	            {
  1245	                s.Outcome = "blocked";
  1246	                throw new CeremonyBlockedException(name, p.Rc);
  1247	            }
  1248	            if (expectation == "success")
  1249	            {
  1250	                // ردُّ خطأٍ من الـTPM كاملُ البنيةِ (رأسٌ فقط) — نتيجةٌ معلومةٌ مسمّاةٌ (SOL413-05): الأدلّةُ
  1251	                // القرائيّةُ تُستنبَطُ من هذا النوعِ وحدَه؛ الردُّ المشوَّهُ فشلٌ آخرُ لا دليلَ غيابٍ.
